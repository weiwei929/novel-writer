## 📋 问题检测方案

### 1. API一致性检测器

#### 检测脚本：`scripts/check-api-consistency.js`
```javascript
const fs = require('fs');
const path = require('path');

// 检测API路由一致性
function checkApiConsistency() {
  const backendDir = path.join(__dirname, '../backend/src');
  const frontendDir = path.join(__dirname, '../frontend/src');
  
  // 1. 检查路由重复定义
  const indexContent = fs.readFileSync(path.join(backendDir, 'index.ts'), 'utf8');
  const routeMatches = indexContent.match(/app\.(get|post|put|delete)\(['"`]([^'"`]+)['"`]/g);
  const routes = routeMatches ? routeMatches.map(match => match.split(/[('"]+/)[1]) : [];
  const duplicates = routes.filter((route, index) => routes.indexOf(route) !== index);
  
  if (duplicates.length > 0) {
    console.error('发现重复路由定义:', duplicates);
  }
  
  // 2. 检查API前缀一致性
  const apiRoutes = routes.filter(route => route.startsWith('/api'));
  const inconsistentPrefixes = [...new Set(apiRoutes.map(route => {
    const parts = route.split('/');
    return parts[1]; // 提取 'api' 或 'api/v1'
  }))];
  
  if (inconsistentPrefixes.length > 1) {
    console.error('API前缀不一致:', inconsistentPrefixes);
  }
  
  // 3. 检查前端API调用与后端路由匹配
  const apiServiceContent = fs.readFileSync(path.join(frontendDir, 'services/api.ts'), 'utf8');
  const frontendApiCalls = apiServiceContent.match(/api\.(get|post|put|delete)\(['"`]([^'"`]+)['"`]/g);
  const frontendRoutes = frontendApiCalls ? frontendApiCalls.map(match => match.split(/[('"]+/)[1]) : [];
  
  // 检查未匹配的路由
  const unmatchedRoutes = frontendRoutes.filter(route => {
    return !routes.some(backendRoute => backendRoute.endsWith(route));
  });
  
  if (unmatchedRoutes.length > 0) {
    console.error('前端调用了不存在的API:', unmatchedRoutes);
  }
  
  return {
    duplicateRoutes: duplicates,
    inconsistentPrefixes,
    unmatchedRoutes
  };
}

module.exports = { checkApiConsistency };
```

### 2. 类型一致性检测器

#### 检测脚本：`scripts/check-type-consistency.js`
```javascript
const fs = require('fs');
const path = require('path');

// 检测前后端类型定义一致性
function checkTypeConsistency() {
  const backendTypesPath = path.join(__dirname, '../backend/src/types/index.ts');
  const frontendTypesPath = path.join(__dirname, '../frontend/src/services/api.ts');
  
  const backendTypes = extractInterfaces(fs.readFileSync(backendTypesPath, 'utf8'));
  const frontendTypes = extractInterfaces(fs.readFileSync(frontendTypesPath, 'utf8'));
  
  const inconsistencies = [];
  
  // 检查共同接口的字段差异
  ['Collection', 'Project', 'Chapter'].forEach(interfaceName => {
    const backendInterface = backendTypes[interfaceName];
    const frontendInterface = frontendTypes[interfaceName];
    
    if (!backendInterface || !frontendInterface) return;
    
    const backendFields = Object.keys(backendInterface);
    const frontendFields = Object.keys(frontendInterface);
    
    // 后端有但前端没有的字段
    const missingInFrontend = backendFields.filter(field => !frontendFields.includes(field));
    if (missingInFrontend.length > 0) {
      inconsistencies.push({
        interface: interfaceName,
        type: 'missing-in-frontend',
        fields: missingInFrontend
      });
    }
    
    // 前端有但后端没有的字段
    const missingInBackend = frontendFields.filter(field => !backendFields.includes(field));
    if (missingInBackend.length > 0) {
      inconsistencies.push({
        interface: interfaceName,
        type: 'missing-in-backend',
        fields: missingInBackend
      });
    }
    
    // 类型不匹配的字段
    const typeMismatches = [];
    backendFields.forEach(field => {
      if (frontendFields.includes(field)) {
        if (backendInterface[field] !== frontendInterface[field]) {
          typeMismatches.push({
            field,
            backendType: backendInterface[field],
            frontendType: frontendInterface[field]
          });
        }
      }
    });
    
    if (typeMismatches.length > 0) {
      inconsistencies.push({
        interface: interfaceName,
        type: 'type-mismatch',
        fields: typeMismatches
      });
    }
  });
  
  return inconsistencies;
}

// 从TypeScript代码中提取接口定义
function extractInterfaces(content) {
  const interfaces = {};
  const interfaceRegex = /export\s+interface\s+(\w+)\s*{([^}]+)}/g;
  let match;
  
  while ((match = interfaceRegex.exec(content)) !== null) {
    const name = match[1];
    const body = match[2];
    
    const fields = {};
    const fieldRegex = /(\w+)(\??):\s*([^;]+)/g;
    let fieldMatch;
    
    while ((fieldMatch = fieldRegex.exec(body)) !== null) {
      const fieldName = fieldMatch[1];
      const fieldType = fieldMatch[2].trim();
      fields[fieldName] = fieldType;
    }
    
    interfaces[name] = fields;
  }
  
  return interfaces;
}

module.exports = { checkTypeConsistency };
```

### 3. 响应格式一致性检测器

#### 检测脚本：`scripts/check-response-consistency.js`
```javascript
const fs = require('fs');
const path = require('path');

// 检测API响应格式一致性
function checkResponseConsistency() {
  const backendDir = path.join(__dirname, '../backend/src/routes');
  const frontendDir = path.join(__dirname, '../frontend/src/services');
  
  const routeFiles = fs.readdirSync(backendDir)
    .filter(file => file.endsWith('.ts') && file !== 'index.ts');
  
  const inconsistencies = [];
  
  routeFiles.forEach(file => {
    const content = fs.readFileSync(path.join(backendDir, file), 'utf8');
    
    // 检查路由中的响应格式
    const routeHandlers = content.match(/router\.(get|post|put|delete)\([^)]+\)\s*async\s*\([^)]+\)\s*=>\s*{([^}]+})/gs);
    
    if (routeHandlers) {
      routeHandlers.forEach(handler => {
        // 检查是否使用标准响应格式
        const usesStandardResponse = handler.includes('success:') && 
          (handler.includes('data:') || handler.includes('error:'));
        
        if (!usesStandardResponse) {
          inconsistencies.push({
            file,
            type: 'non-standard-response',
            handler: handler.substring(0, 100) + '...'
          });
        }
      });
    }
  });
  
  // 检查前端API调用中的响应处理
  const apiServiceContent = fs.readFileSync(path.join(frontendDir, 'api.ts'), 'utf8');
  const apiCalls = apiServiceContent.match(/async\s+\w+\([^)]+\):\s*Promise<[^>]+>\s*{([^}]+})/gs);
  
  if (apiCalls) {
    apiCalls.forEach(call => {
      // 检查是否正确处理ApiResponse格式
      const checksSuccessField = call.includes('response.data.success');
      const checksDataField = call.includes('response.data.data');
      const checksErrorField = call.includes('response.data.error');
      
      if (!checksSuccessField || (!checksDataField && !checksErrorField)) {
        inconsistencies.push({
          file: 'api.ts',
          type: 'incorrect-response-handling',
          handler: call.substring(0, 100) + '...'
        });
      }
    });
  }
  
  return inconsistencies;
}

module.exports = { checkResponseConsistency };
```

### 1. API路由统一修复方案

#### 修复脚本：`scripts/fix-api-routes.js`
```javascript
const fs = require('fs');
const path = require('path');

// 修复API路由一致性问题
function fixApiRoutes() {
  const backendDir = path.join(__dirname, '../backend/src');
  const indexFile = path.join(backendDir, 'index.ts');
  
  // 1. 创建备份
  const backupFile = indexFile + '.backup';
  fs.copyFileSync(indexFile, backupFile);
  console.log('已创建备份:', backupFile);
  
  // 2. 读取当前内容
  let content = fs.readFileSync(indexFile, 'utf8');
  
  // 3. 删除重复的健康检查路由
  const healthRouteRegex = /app\.get\(['"`]\/health['"`],[^}]+}\)/g;
  const matches = content.match(healthRouteRegex);
  if (matches && matches.length > 1) {
    // 保留第一个，删除其余的
    content = content.replace(healthRouteRegex, (match, offset, fullString) => {
      // 如果是第一个匹配项，保留
      if (fullString.substring(0, offset).includes('app.get')) {
        return match;
      }
      // 否则替换为空字符串
      return '';
    });
  }
  
  // 4. 统一API前缀为 /api/v1
  content = content.replace(/app\.use\(['"`]\/api['"`]/g, "app.use('/api/v1'");
  
  // 5. 修复路由挂载顺序
  const routeMounts = [
    "app.use('/api/v1/collections', collectionsRouter)",
    "app.use('/api/v1/projects', projectsRouter)",
    "app.use('/api/v1/chapters', chaptersRouter)",
    "app.use('/api/v1/stats', statsRouter)",
    "app.use('/api/v1/ai', grokRouter)"
  ];
  
  // 查找路由挂载部分
  const routesSectionRegex = /\/\/ API路由[\s\S]*?(?=\n\n\/\/)/;
  const routesSection = content.match(routesSectionRegex);
  
  if (routesSection) {
    const newRoutesSection = `// API路由\n${routeMounts.join('\n')}`;
    content = content.replace(routesSectionRegex, newRoutesSection);
  }
  
  // 6. 写入修复后的内容
  fs.writeFileSync(indexFile, content);
  console.log('API路由已修复');
  
  return {
    fixed: true,
    backupFile
  };
}

module.exports = { fixApiRoutes };
```

#### 修复脚本：`scripts/fix-response-consistency.js`
```javascript
const fs = require('fs');
const path = require('path');

// 修复API响应格式一致性问题
function fixResponseConsistency() {
  const backendDir = path.join(__dirname, '../backend/src/routes');
  const frontendDir = path.join(__dirname, '../frontend/src/services');
  
  // 1. 创建标准响应格式工具
  const responseUtilPath = path.join(__dirname, '../backend/src/utils/response.ts');
  const responseUtilContent = `
import { Response } from 'express';
import { ApiResponse } from '../types';

export const sendSuccess = <T>(res: Response, data: T, metadata = {}) => {
  const response: ApiResponse<T> = {
    success: true,
    data,
    metadata: {
      timestamp: new Date().toISOString(),
      version: '1.0.0',
      ...metadata
    }
  };
  
  res.json(response);
};

export const sendError = (res: Response, code: string, message: string, statusCode = 500) => {
  const response: ApiResponse = {
    success: false,
    error: { code, message }
  };
  
  res.status(statusCode).json(response);
};
`;
  
  // 确保目录存在
  const utilsDir = path.dirname(responseUtilPath);
  if (!fs.existsSync(utilsDir)) {
    fs.mkdirSync(utilsDir, { recursive: true });
  }
  
  // 写入响应工具
  fs.writeFileSync(responseUtilPath, responseUtilContent);
  console.log('已创建响应格式工具');
  
  // 2. 修复后端路由中的响应格式
  const routeFiles = fs.readdirSync(backendDir)
    .filter(file => file.endsWith('.ts') && file !== 'index.ts');
  
  routeFiles.forEach(file => {
    const filePath = path.join(backendDir, file);
    
    // 创建备份
    fs.copyFileSync(filePath, filePath + '.backup');
    
    // 读取内容
    let content = fs.readFileSync(filePath, 'utf8');
    
    // 添加导入
    if (!content.includes('import { sendSuccess, sendError }')) {
      content = content.replace(
        "import { ApiResponse",
        "import { sendSuccess, sendError } from '../utils/response'\nimport { ApiResponse"
      );
    }
    
    // 替换成功响应
    content = content.replace(
      /res\.json\(\{\s*success:\s*true,\s*data:\s*([^}]+)\s*\}\)/g,
      'sendSuccess(res, $1)'
    );
    
    // 替换错误响应
    content = content.replace(
      /res\.status\((\d+)\)\.json\(\{\s*success:\s*false,\s*error:\s*\{\s*code:\s*([^,]+),\s*message:\s*([^}]+)\s*\}\s*\}\)/g,
      'sendError(res, $1, $2, $1)'
    );
    
    // 写入修复后的内容
    fs.writeFileSync(filePath, content);
    console.log(`已修复 ${file} 中的响应格式`);
  });
  
  // 3. 修复前端API服务中的响应处理
  const apiServicePath = path.join(frontendDir, 'api.ts');
  
  // 创建备份
  fs.copyFileSync(apiServicePath, apiServicePath + '.backup');
  
  // 读取内容
  let apiContent = fs.readFileSync(apiServicePath, 'utf8');
  
  // 添加响应处理工具
  const responseHandlerContent = `
// 处理API响应
const handleResponse = (response) => {
  if (response.data && typeof response.data === 'object' && 'success' in response.data) {
    return response.data;
  }
  
  // 兼容旧格式
  return {
    success: true,
    data: response.data
  };
};
`;
  
  // 在文件开头添加响应处理工具
  apiContent = apiContent.replace(
    "const api = axios.create({",
    responseHandlerContent + "\nconst api = axios.create({"
  );
  
  // 修复API调用中的响应处理
  apiContent = apiContent.replace(
    /const response = await api\.(get|post|put|delete)\([^)]+\);\s*(?:if|return)\s*response\.data\.(success|data|error)/g,
    (match, method, checkField) => {
      return `const response = await api.${method}($&{});
  const apiResponse = handleResponse(response);
  if (apiResponse.${checkField})`;
    }
  );
  
  // 写入修复后的内容
  fs.writeFileSync(apiServicePath, apiContent);
  console.log('已修复前端API服务中的响应处理');
  
  return {
    fixed: true,
    backendFiles: routeFiles,
    frontendFile: 'api.ts'
  };
}

module.exports = { fixResponseConsistency };
```

```javascript
const fs = require('fs');
const path = require('path');

// 修复章节API集成问题
function fixChapterApiIntegration() {
  const frontendDir = path.join(__dirname, '../frontend/src');
  const editorPagePath = path.join(frontendDir, 'pages/EditorPage.tsx');
  const apiServicePath = path.join(frontendDir, 'services/api.ts');

  fs.copyFileSync(editorPagePath, editorPagePath + '.backup');
  fs.copyFileSync(apiServicePath, apiServicePath + '.backup');

  let apiContent = fs.readFileSync(apiServicePath, 'utf8');
  
  // 检查是否已有章节API
  if (!apiContent.includes('chaptersApi')) {
    // 章节接口定义

// 章节相关类型定义
export interface Chapter {
  id: string
  title: string
  content: string
  projectId: string
  order: number
  wordCount: number
  status: 'draft' | 'writing' | 'completed' | 'published'
  createdAt: string
  updatedAt: string
}

export interface CreateChapterData {
  title: string
  content?: string
  projectId: string
  order?: number
}
`;
    
    // 章节API实现
    const chapterApiImplementation = `
// 章节 API 服务
export const chaptersApi = {
  // 获取项目的所有章节
  async getAll(projectId: string): Promise<Chapter[]> {
    const response = await api.get(\`/projects/\${projectId}/chapters\`)
    if (response.data.success) {
      return response.data.data || []
    } else {
      throw new Error(response.data.error?.message || '获取章节失败')
    }
  },

  // 根据ID获取章节
  async getById(id: string): Promise<Chapter> {
    const response = await api.get(\`/chapters/\${id}\`)
    if (response.data.success) {
      return response.data.data
    } else {
      throw new Error(response.data.error?.message || '获取章节失败')
    }
  },

  // 创建章节
  async create(data: CreateChapterData): Promise<Chapter> {
    const response = await api.post('/chapters', data)
    if (response.data.success) {
      return response.data.data
    } else {
      throw new Error(response.data.error?.message || '创建章节失败')
    }
  },

  // 更新章节
  async update(id: string, data: Partial<Chapter>): Promise<Chapter> {
    const response = await api.put(\`/chapters/\${id}\`, data)
    if (response.data.success) {
      return response.data.data
    } else {
      throw new Error(response.data.error?.message || '更新章节失败')
    }
  },

  // 删除章节
  async delete(id: string): Promise<void> {
    const response = await api.delete(\`/chapters/\${id}\`)
    if (!response.data.success) {
      throw new Error(response.data.error?.message || '删除章节失败')
    }
  },
}
`;
    
    // 在文件末尾添加章节API
    apiContent += chapterInterface + chapterApiImplementation;
    
    // 写入修复后的内容
    fs.writeFileSync(apiServicePath, apiContent);
    console.log('已添加章节API到前端服务');
  }
  
  // 3. 修复编辑器页面中的章节加载和保存
  let editorContent = fs.readFileSync(editorPagePath, 'utf8');
  
  // 添加章节API导入
  editorContent = editorContent.replace(
    "import { projectsApi, Project } from '../services/api'",
    "import { projectsApi, chaptersApi, Project, Chapter } from '../services/api'"
  );
  
  // 实现章节加载功能
  const loadChapterImplementation = `
  const loadChapter = async () => {
    if (!chapterId) return
    
    try {
      setLoading(true)
      const chapter = await chaptersApi.getById(chapterId)
      setContent(chapter.content)
    } catch (err) {
      setError('加载章节失败')
      console.error('Error loading chapter:', err)
    } finally {
      setLoading(false)
    }
  }
`;
  
  // 替换TODO注释
  editorContent = editorContent.replace(
    "// TODO: 加载章节内容",
    loadChapterImplementation
  );
  
  // 实现章节保存功能
  const saveChapterImplementation = `
  const handleSave = async (content: string) => {
    if (!chapterId) {
      // 创建新章节
      try {
        const newChapter = await chaptersApi.create({
          title: '新章节',
          content,
          projectId
        })
        // 更新URL到新章节
        navigate(\`/editor/\${projectId}/\${newChapter.id}\`)
      } catch (err) {
        setError('保存失败')
        console.error('Error creating chapter:', err)
      }
    } else {
      // 更新现有章节
      try {
        await chaptersApi.update(chapterId, { content })
        setLastSaved(new Date())
      } catch (err) {
        setError('保存失败')
        console.error('Error saving chapter:', err)
      }
    }
  }
`;
  
  // 替换TODO注释
  editorContent = editorContent.replace(
    "// TODO: 实现保存逻辑",
    saveChapterImplementation
  );
  
  // 更新useEffect以调用loadChapter
  editorContent = editorContent.replace(
    "useEffect(() => {\n    if (projectId) {\n      loadProject()\n    } else {\n      setLoading(false)\n    }\n  }, [projectId])",
```

```javascript
function checkTypeConsistency() {
  const issues = [];
  
  // 1. 检查前后端类型定义一致性
  const backendTypesPath = path.join(__dirname, '../backend/src/types/index.ts');
  const frontendTypesPath = path.join(__dirname, '../frontend/src/services/api.ts');
  
  const backendTypes = extractInterfaces(fs.readFileSync(backendTypesPath, 'utf8'));
  const frontendTypes = extractInterfaces(fs.readFileSync(frontendTypesPath, 'utf8'));
  
  ['Collection', 'Project', 'Chapter'].forEach(interfaceName => {
    const backendInterface = backendTypes[interfaceName];
    const frontendInterface = frontendTypes[interfaceName];
    
    if (!backendInterface || !frontendInterface) {
      issues.push({
        type: 'missing-interface',
        severity: 'high',
        location: !backendInterface ? backendTypesPath : frontendTypesPath,
        description: `${interfaceName} 接口在${!backendInterface ? '后端' : '前端'}未定义`,
        suggestion: `在${!backendInterface ? '后端' : '前端'}添加 ${interfaceName} 接口定义`
      });
      return;
    }
    
    // 检查字段差异
    const backendFields = Object.keys(backendInterface);
    const frontendFields = Object.keys(frontendInterface);
    
    // 后端有但前端没有的字段
    const missingInFrontend = backendFields.filter(field => !frontendFields.includes(field));
    if (missingInFrontend.length > 0) {
      issues.push({
        type: 'missing-frontend-fields',
        severity: 'medium',
        location: frontendTypesPath,
        description: `${interfaceName} 接口在前端缺少字段: ${missingInFrontend.join(', ')}`,
        suggestion: `在前端 ${interfaceName} 接口中添加: ${missingInFrontend.map(f => `${f}: ${backendInterface[f]}`).join(', ')}`
      });
    }
    
    // 前端有但后端没有的字段
    const missingInBackend = frontendFields.filter(field => !backendFields.includes(field));
    if (missingInBackend.length > 0) {
      issues.push({
        type: 'missing-backend-fields',
        severity: 'medium',
        location: backendTypesPath,
        description: `${interfaceName} 接口在后端缺少字段: ${missingInBackend.join(', ')}`,
        suggestion: `在后端 ${interfaceName} 接口中添加: ${missingInBackend.map(f => `${f}: ${frontendInterface[f]}`).join(', ')}`
      });
    }
    
    // 类型不匹配的字段
    const typeMismatches = [];
    backendFields.forEach(field => {
      if (frontendFields.includes(field) && backendInterface[field] !== frontendInterface[field]) {
        typeMismatches.push({
          field,
          backendType: backendInterface[field],
          frontendType: frontendInterface[field]
        });
      }
    });
    
    if (typeMismatches.length > 0) {
      issues.push({
        type: 'type-mismatch',
        severity: 'medium',
        location: `${backendTypesPath}, ${frontendTypesPath}`,
        description: `${interfaceName} 接口字段类型不匹配: ${typeMismatches.map(m => `${m.field} (${m.backendType} vs ${m.frontendType})`).join(', ')}`,
        suggestion: '统一前后端接口字段类型定义'
      });
    }
  });
  
  return issues;
}
```
